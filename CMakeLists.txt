cmake_minimum_required(VERSION 3.20)
project(Escalation VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ESCALATION_BUILD_CLIENT "Build the native OpenGL desktop game" ON)
option(ESCALATION_NULL_PLATFORM "Build GLFW without a desktop backend for render smoke tests" OFF)
include(CTest)
add_library(escalation_core src/world.cpp src/simulation.cpp src/simulation_actions.cpp src/combat.cpp)
target_include_directories(escalation_core PUBLIC include)
target_include_directories(escalation_core SYSTEM PRIVATE third_party/picojson third_party/earcut)
if(MSVC)
 target_compile_options(escalation_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(escalation_core PRIVATE -Wall -Wextra)
endif()
add_executable(escalation_headless src/headless.cpp)
target_link_libraries(escalation_headless PRIVATE escalation_core)
target_compile_definitions(escalation_headless PRIVATE ESCALATION_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
if(BUILD_TESTING)
 add_executable(escalation_tests tests/core_tests.cpp)
 target_link_libraries(escalation_tests PRIVATE escalation_core)
 target_include_directories(escalation_tests SYSTEM PRIVATE third_party/picojson)
 add_executable(escalation_camera_tests tests/camera_tests.cpp src/renderer.cpp src/assets.cpp)
 target_link_libraries(escalation_camera_tests PRIVATE escalation_core)
 target_include_directories(escalation_camera_tests SYSTEM PRIVATE third_party/picojson third_party/stb)
 add_test(NAME projection_roundtrip COMMAND escalation_camera_tests)
 add_test(NAME simulation_invariants COMMAND escalation_tests)
 add_executable(escalation_tactical_tests tests/tactical_tests.cpp)
 target_link_libraries(escalation_tactical_tests PRIVATE escalation_core)
 add_test(NAME tactical_invariants COMMAND escalation_tactical_tests)
 set_tests_properties(tactical_invariants PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
 set_tests_properties(simulation_invariants PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(ESCALATION_BUILD_CLIENT)
 include(FetchContent)
 set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
 set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
 set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
 if(ESCALATION_NULL_PLATFORM)
  set(GLFW_BUILD_X11 OFF CACHE BOOL "" FORCE)
 endif()
 FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
 FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.8 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(glfw imgui)
 add_library(escalation_ui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
 target_include_directories(escalation_ui SYSTEM PUBLIC ${imgui_SOURCE_DIR})
 target_compile_definitions(escalation_ui PRIVATE GLFW_INCLUDE_NONE)
 target_link_libraries(escalation_ui PUBLIC glfw ${CMAKE_DL_LIBS})
 add_executable(escalation src/main.cpp src/renderer.cpp src/assets.cpp)
 target_include_directories(escalation SYSTEM PRIVATE third_party/picojson third_party/stb)
 target_link_libraries(escalation PRIVATE escalation_core escalation_ui)
 target_compile_definitions(escalation PRIVATE ESCALATION_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
 if(MINGW)
  target_link_options(escalation PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(escalation_headless PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS escalation escalation_headless RUNTIME DESTINATION .)
 install(DIRECTORY data DESTINATION .)
 install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
 install(DIRECTORY third_party/licenses DESTINATION third_party)
 if(ESCALATION_NULL_PLATFORM AND BUILD_TESTING)
  add_test(NAME opengl_smoke COMMAND escalation --smoke "${CMAKE_CURRENT_BINARY_DIR}/smoke")
  set_tests_properties(opengl_smoke PROPERTIES ENVIRONMENT "EGL_PLATFORM=surfaceless")
 endif()
endif()
